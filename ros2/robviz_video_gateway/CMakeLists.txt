cmake_minimum_required(VERSION 3.16)
project(robviz_video_gateway)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs
  PATHS /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/cmake/opencv4 NO_DEFAULT_PATH)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-webrtc-1.0 gstreamer-sdp-1.0 libsoup-2.4)
add_library(video_gateway SHARED src/gateway.cpp src/support.cpp)
target_compile_definitions(video_gateway PRIVATE GST_USE_UNSTABLE_API)
target_include_directories(video_gateway PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(video_gateway rclcpp rclcpp_components sensor_msgs std_srvs)
target_link_libraries(video_gateway PkgConfig::MEDIA yaml-cpp nlohmann_json::nlohmann_json ${OpenCV_LIBS})
target_compile_options(video_gateway PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_node(video_gateway PLUGIN "robviz_video_gateway::VideoGateway" EXECUTABLE gateway_node)
install(TARGETS video_gateway ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(video_protocol_check test/test_support.cpp)
  target_link_libraries(video_protocol_check video_gateway PkgConfig::MEDIA yaml-cpp nlohmann_json::nlohmann_json)
  ament_target_dependencies(video_protocol_check rclcpp)
  add_test(NAME video_protocol_check COMMAND video_protocol_check)
  add_executable(synthetic_camera_graph test/synthetic_camera.cpp)
  target_link_libraries(synthetic_camera_graph video_gateway)
  ament_target_dependencies(synthetic_camera_graph rclcpp sensor_msgs std_srvs)
  install(TARGETS synthetic_camera_graph DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
